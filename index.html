/** * ============================================================ * ACTUALIZACIÓN DE EQUIPOS BIOMÉDICOS - Backend (Apps Script) * ============================================================ */ const CONFIG = { SHEET_NAME: 'BBDD', SHEET_NAME_VIDA_UTIL: 'VIDA UTIL', CARPETA_DRIVE_FOTOS: 'Reportes_Equipos_Biomedicos', FILA_ENCABEZADO: 1, CACHE_TTL_SEGUNDOS: 21600, }; // Columnas de la pestaña "VIDA UTIL" (letra -> índice base 1) const COL_VIDA_UTIL = { DENOMINACION: 3, // C VIDA_UTIL: 9, // I }; const COL = { CODIGO_PATRIMONIAL: 1, CODIGO_BARRAS: 2, DESCRIPCION: 3, SEDE: 4, VIGENCIA: 5, MARCA: 6, MODELO: 7, SERIE: 8, SERVICIO: 9, ESTADO: 10, }; const COL_NUEVAS = { FECHA_REVISION: 'FECHA_ULTIMA_REVISION', TECNICO: 'TECNICO_RESPONSABLE', FOTO_URL: 'URL_FOTO_REPORTE', }; function getHoja_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(CONFIG.SHEET_NAME); if (!sheet) throw new Error('No se encontró la pestaña "' + CONFIG.SHEET_NAME + '"'); return sheet; } function inicializarColumnas() { const sheet = getHoja_(); const lastCol = sheet.getLastColumn(); const encabezados = sheet.getRange(CONFIG.FILA_ENCABEZADO, 1, 1, lastCol).getValues()[0]; const nombresNuevos = [COL_NUEVAS.FECHA_REVISION, COL_NUEVAS.TECNICO, COL_NUEVAS.FOTO_URL]; let col = lastCol; nombresNuevos.forEach(function (nombre) { if (encabezados.indexOf(nombre) === -1) { col++; sheet.getRange(CONFIG.FILA_ENCABEZADO, col).setValue(nombre); } }); SpreadsheetApp.getUi().alert('Columnas creadas/verificadas correctamente.'); } function indiceColumnaNueva_(sheet, nombreColumna) { const lastCol = sheet.getLastColumn(); const encabezados = sheet.getRange(CONFIG.FILA_ENCABEZADO, 1, 1, lastCol).getValues()[0]; const idx = encabezados.indexOf(nombreColumna); if (idx === -1) throw new Error('Falta la columna "' + nombreColumna + '". Ejecuta inicializarColumnas() primero.'); return idx + 1; } /* ================= CACHÉ ================= */ const CACHE_PREFIJO = 'indice_v2'; // Posiciones dentro de cada arreglo del índice (más liviano que usar objetos) const IDX = { FILA: 0, COD_PAT: 1, COD_BAR: 2, DESC: 3, SEDE: 4, VIGENCIA: 5, MARCA: 6, MODELO: 7, SERIE: 8, SERVICIO: 9, ESTADO: 10, }; function obtenerIndice_() { const cache = CacheService.getScriptCache(); const numChunks = cache.get(CACHE_PREFIJO + '_n'); if (numChunks) { const claves = []; for (let i = 0; i < Number(numChunks); i++) claves.push(CACHE_PREFIJO + '_' + i); const partes = cache.getAll(claves); let completo = true; let json = ''; for (let i = 0; i < claves.length; i++) { if (!partes[claves[i]]) { completo = false; break; } json += partes[claves[i]]; } if (completo) { try { return JSON.parse(json); } catch (e) {} } } const sheet = getHoja_(); const lastRow = sheet.getLastRow(); if (lastRow < 2) return []; const datos = sheet.getRange(2, 1, lastRow - 1, 10).getValues(); const indice = []; for (let i = 0; i < datos.length; i++) { const fila = datos[i]; if (fila[COL.DESCRIPCION - 1]) { indice.push([ i + 2, fila[COL.CODIGO_PATRIMONIAL - 1], fila[COL.CODIGO_BARRAS - 1], fila[COL.DESCRIPCION - 1], fila[COL.SEDE - 1], fila[COL.VIGENCIA - 1], fila[COL.MARCA - 1], fila[COL.MODELO - 1], fila[COL.SERIE - 1], fila[COL.SERVICIO - 1], fila[COL.ESTADO - 1], ]); } } guardarIndiceEnCache_(indice); return indice; } function guardarIndiceEnCache_(indice) { const cache = CacheService.getScriptCache(); const json = JSON.stringify(indice); const TAM_CHUNK = 90000; const chunks = []; for (let i = 0; i < json.length; i += TAM_CHUNK) { chunks.push(json.substring(i, i + TAM_CHUNK)); } if (chunks.length > 300) return; // seguro de tamaño, no debería alcanzarse const paquete = {}; chunks.forEach(function (c, i) { paquete[CACHE_PREFIJO + '_' + i] = c; }); paquete[CACHE_PREFIJO + '_n'] = String(chunks.length); cache.putAll(paquete, CONFIG.CACHE_TTL_SEGUNDOS); } function invalidarCache_() { CacheService.getScriptCache().remove(CACHE_PREFIJO + '_n'); } /* ================= VIDA ÚTIL (pestaña "VIDA UTIL") ================= */ const CACHE_VIDA_UTIL = 'vida_util_v1'; /** Devuelve un objeto { DENOMINACION_EN_MAYUSCULAS: vidaUtilEnAños } */ function obtenerMapaVidaUtil_() { const cache = CacheService.getScriptCache(); const guardado = cache.get(CACHE_VIDA_UTIL); if (guardado) { try { return JSON.parse(guardado); } catch (e) {} } const ss = SpreadsheetApp.getActiveSpreadsheet(); const hoja = ss.getSheetByName(CONFIG.SHEET_NAME_VIDA_UTIL); if (!hoja) return {}; const lastRow = hoja.getLastRow(); if (lastRow < 2) return {}; const datos = hoja.getRange(2, 1, lastRow - 1, COL_VIDA_UTIL.VIDA_UTIL).getValues(); const mapa = {}; datos.forEach(function (fila) { const denominacion = (fila[COL_VIDA_UTIL.DENOMINACION - 1] || '').toString().trim().toUpperCase(); const vidaUtil = fila[COL_VIDA_UTIL.VIDA_UTIL - 1]; if (denominacion && vidaUtil !== '' && vidaUtil !== null) { mapa[denominacion] = vidaUtil; } }); try { cache.put(CACHE_VIDA_UTIL, JSON.stringify(mapa), CONFIG.CACHE_TTL_SEGUNDOS); } catch (e) { // si el catálogo es muy grande para cachear, simplemente no se cachea } return mapa; } /** Busca la vida útil de un equipo comparando su descripción contra el catálogo */ function buscarVidaUtil_(descripcionEquipo) { if (!descripcionEquipo) return null; const mapa = obtenerMapaVidaUtil_(); const desc = descripcionEquipo.toString().trim().toUpperCase(); // 1. Coincidencia exacta if (mapa[desc] !== undefined) return mapa[desc]; // 2. Coincidencia parcial (la descripción contiene la denominación, o viceversa) let mejorCoincidencia = null; let mejorLongitud = 0; Object.keys(mapa).forEach(function (denominacion) { if (desc.indexOf(denominacion) !== -1 || denominacion.indexOf(desc) !== -1) { if (denominacion.length > mejorLongitud) { mejorLongitud = denominacion.length; mejorCoincidencia = mapa[denominacion]; } } }); return mejorCoincidencia; } /* ================= ENDPOINTS ================= */ function doGet(e) { try { const accion = e.parameter.accion; let resultado; if (accion === 'buscarPorCodigo') { resultado = buscarPorCodigo_(e.parameter.codigo); } else if (accion === 'listarSedes') { resultado = listarSedes_(); } else if (accion === 'listarDependencias') { resultado = listarServicios_(e.parameter.sede); } else if (accion === 'listarEquipos') { resultado = listarEquipos_(e.parameter.sede, e.parameter.dependencia); } else if (accion === 'limpiarCache') { invalidarCache_(); resultado = { ok: true, mensaje: 'Caché limpiada.' }; } else { resultado = { error: 'Acción no reconocida' }; } return respuestaJson_(resultado); } catch (err) { return respuestaJson_({ error: err.message }); } } function doPost(e) { try { const datos = JSON.parse(e.postData.contents); let resultado; if (datos.accion === 'registrar') { resultado = registrarEquipo_(datos); } else { resultado = actualizarEstado_(datos); } return respuestaJson_(resultado); } catch (err) { return respuestaJson_({ error: err.message }); } } function respuestaJson_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } /* ================= BÚSQUEDA ================= */ function buscarPorCodigo_(codigo) { if (!codigo) return { error: 'Código vacío' }; const codigoStr = codigo.toString().trim(); const indice = obtenerIndice_(); const encontrado = indice.find(function (eq) { return (eq[IDX.COD_BAR] || '').toString().trim() === codigoStr || (eq[IDX.COD_PAT] || '').toString().trim() === codigoStr; }); if (encontrado) return equipoAObjetoRespuesta_(encontrado); return { error: 'No se encontró ningún equipo con ese código en la base de datos.', noEncontrado: true, codigo: codigoStr }; } function listarSedes_() { const indice = obtenerIndice_(); const sedes = new Set(); indice.forEach(function (eq) { const sede = (eq[IDX.SEDE] || '').toString().trim(); if (sede) sedes.add(sede); }); return { sedes: Array.from(sedes).sort() }; } function listarServicios_(sede) { const indice = obtenerIndice_(); const servicios = new Set(); indice.forEach(function (eq) { if ((eq[IDX.SEDE] || '').toString().trim() === sede) { const servicio = (eq[IDX.SERVICIO] || '').toString().trim(); if (servicio) servicios.add(servicio); } }); return { dependencias: Array.from(servicios).sort() }; } function listarEquipos_(sede, servicio) { const indice = obtenerIndice_(); const equipos = indice .filter(function (eq) { const coincideSede = (eq[IDX.SEDE] || '').toString().trim() === sede; const coincideServicio = !servicio || (eq[IDX.SERVICIO] || '').toString().trim() === servicio; return coincideSede && coincideServicio; }) .map(equipoAObjetoRespuesta_); return { equipos: equipos }; } function equipoAObjetoRespuesta_(eq) { return { fila: eq[IDX.FILA], codigoPatrimonial: eq[IDX.COD_PAT], codigoBarras: eq[IDX.COD_BAR], descripcion: eq[IDX.DESC], sede: eq[IDX.SEDE], vigencia: eq[IDX.VIGENCIA], dependencia: eq[IDX.SERVICIO], marca: eq[IDX.MARCA], modelo: eq[IDX.MODELO], serie: eq[IDX.SERIE], ubicacion: eq[IDX.SERVICIO], estadoActual: eq[IDX.ESTADO], vidaUtil: buscarVidaUtil_(eq[IDX.DESC]), }; } /* ================= ACTUALIZACIÓN ================= */ function actualizarEstado_(datos) { const lock = LockService.getScriptLock(); lock.waitLock(15000); try { const sheet = getHoja_(); const fila = Number(datos.fila); if (!fila) throw new Error('Fila inválida'); let fotoUrl = ''; if (datos.fotoBase64) { fotoUrl = guardarFoto_(datos.fotoBase64, datos.codigoBarras || datos.codigoPatrimonial || fila); } const colFecha = indiceColumnaNueva_(sheet, COL_NUEVAS.FECHA_REVISION); const colTecnico = indiceColumnaNueva_(sheet, COL_NUEVAS.TECNICO); const colFoto = indiceColumnaNueva_(sheet, COL_NUEVAS.FOTO_URL); sheet.getRange(fila, COL.ESTADO).setValue(datos.estado); sheet.getRange(fila, colFecha).setValue(new Date()); sheet.getRange(fila, colTecnico).setValue(datos.tecnico || ''); if (fotoUrl) sheet.getRange(fila, colFoto).setValue(fotoUrl); // Campos editables: Marca, Modelo, Serie, Sede, Servicio (solo si vienen en la petición) if (typeof datos.marca === 'string') sheet.getRange(fila, COL.MARCA).setValue(datos.marca); if (typeof datos.modelo === 'string') sheet.getRange(fila, COL.MODELO).setValue(datos.modelo); if (typeof datos.serie === 'string') sheet.getRange(fila, COL.SERIE).setValue(datos.serie); if (typeof datos.sede === 'string' && datos.sede) sheet.getRange(fila, COL.SEDE).setValue(datos.sede); if (typeof datos.servicio === 'string') sheet.getRange(fila, COL.SERVICIO).setValue(datos.servicio); invalidarCache_(); return { exito: true, fotoUrl: fotoUrl }; } finally { lock.releaseLock(); } } /** Registra un equipo nuevo cuando su código no existe en la base */ function registrarEquipo_(datos) { const lock = LockService.getScriptLock(); lock.waitLock(15000); try { const sheet = getHoja_(); const lastCol = sheet.getLastColumn(); const fila = new Array(lastCol).fill(''); fila[COL.CODIGO_PATRIMONIAL - 1] = datos.codigoPatrimonial || ''; fila[COL.CODIGO_BARRAS - 1] = datos.codigoBarras || ''; fila[COL.DESCRIPCION - 1] = datos.descripcion || ''; fila[COL.SEDE - 1] = datos.sede || ''; fila[COL.MARCA - 1] = datos.marca || ''; fila[COL.MODELO - 1] = datos.modelo || ''; fila[COL.SERIE - 1] = datos.serie || ''; fila[COL.SERVICIO - 1] = datos.servicio || ''; fila[COL.ESTADO - 1] = datos.estado || 'OPERATIVO'; const colFecha = indiceColumnaNueva_(sheet, COL_NUEVAS.FECHA_REVISION); const colTecnico = indiceColumnaNueva_(sheet, COL_NUEVAS.TECNICO); fila[colFecha - 1] = new Date(); fila[colTecnico - 1] = datos.tecnico || ''; sheet.appendRow(fila); invalidarCache_(); return { exito: true, fila: sheet.getLastRow() }; } finally { lock.releaseLock(); } } function guardarFoto_(base64, nombreBase) { const carpetas = DriveApp.getFoldersByName(CONFIG.CARPETA_DRIVE_FOTOS); const carpeta = carpetas.hasNext() ? carpetas.next() : DriveApp.createFolder(CONFIG.CARPETA_DRIVE_FOTOS); const partes = base64.split(','); const contenido = partes.length > 1 ? partes[1] : partes[0]; const blob = Utilities.newBlob(Utilities.base64Decode(contenido), 'image/jpeg', 'reporte_' + nombreBase + '_' + new Date().getTime() + '.jpg'); const archivo = carpeta.createFile(blob); archivo.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return archivo.getUrl(); }
