/** * ============================================================ * ACTUALIZACIÓN DE EQUIPOS BIOMÉDICOS - Backend (Apps Script) * ============================================================ * * La hoja "BBDD" tiene una columna REAL con la fecha de alta del * equipo (columna E). La vigencia se calcula en vivo desde esa fecha. * * Layout de columnas fijas en "BBDD": * A: Código Patrimonial * B: Código de Barras * C: Descripción * D: Sede / Establecimiento de Salud * E: Fecha de Alta del Equipo <-- fuente de verdad de la vigencia * F: Vigencia del Equipo (texto "X AÑOS Y MESES") — la escribe * esta app cada vez que se registra o actualiza el equipo, * calculada en vivo a partir de E. * G: Marca * H: Modelo * I: Serie * J: Servicio * K: Estado * * Las columnas de metadata (fecha de revisión, técnico, observaciones) * se siguen creando dinámicamente al final de la hoja. * * ------------------------------------------------------------ * NOTA DE RENDIMIENTO (por qué esta versión es más rápida): * Con ~19,700 filas, reconstruir el índice completo desde la hoja es * lo más caro que hace este script. Antes, CUALQUIER actualización de * un equipo borraba por completo la caché general, así que la * siguiente búsqueda o consulta de porcentaje forzaba esa * reconstrucción completa. Ahora: * - Buscar por código usa un índice aparte y liviano (código -> fila) * y lee esa UNA fila directo de la hoja: no toca el índice pesado. * - El porcentaje vive en su propia caché chica que se ajusta en * memoria (+1) en vez de recalcularse leyendo toda la hoja. * - Actualizar un equipo ya no invalida el índice pesado (se deja * vencer solo, cada 10 min) y escribe sus columnas en bloque en * vez de una por una. * - Registrar un equipo SÍ sigue invalidando el índice pesado y el * de códigos, porque reordena toda la hoja y los números de fila * de todos los equipos pueden cambiar. * ------------------------------------------------------------ */ const CONFIG = { SHEET_NAME: 'BBDD', SHEET_NAME_VIDA_UTIL: 'VIDA UTIL', FILA_ENCABEZADO: 1, CACHE_TTL_SEGUNDOS: 600, // 10 minutos: se refresca sola; el botón "Actualizar sistema" también la limpia al toque }; // Columnas de la pestaña "VIDA UTIL" (letra -> índice base 1) const COL_VIDA_UTIL = { DENOMINACION: 3, // C VIDA_UTIL: 9, // I }; const COL = { CODIGO_PATRIMONIAL: 1, CODIGO_BARRAS: 2, DESCRIPCION: 3, SEDE: 4, FECHA_ALTA: 5, // E — fecha real de alta del equipo VIGENCIA: 6, // F — texto "X AÑOS Y MESES", esta app SÍ la escribe (no hay fórmula real en la hoja) MARCA: 7, MODELO: 8, SERIE: 9, SERVICIO: 10, ESTADO: 11, }; const COL_NUEVAS = { FECHA_REVISION: 'FECHA_ULTIMA_REVISION', TECNICO: 'TECNICO_RESPONSABLE', OBSERVACIONES: 'OBSERVACIONES', }; // Cuántos milisegundos tiene un año (usa 365.25 para compensar años bisiestos) const MS_POR_ANIO = 365.25 * 24 * 60 * 60 * 1000; function getHoja_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(CONFIG.SHEET_NAME); if (!sheet) throw new Error('No se encontró la pestaña "' + CONFIG.SHEET_NAME + '"'); return sheet; } /** Crea un menú en la hoja de cálculo para poder ordenar manualmente cuando se quiera, * además del ordenamiento automático que ya ocurre al registrar un equipo nuevo. */ function onOpen() { SpreadsheetApp.getUi() .createMenu('⚙️ Equipos Biomédicos') .addItem('Ordenar hoja ahora (Sede → Servicio → Descripción)', 'ordenarHojaManual_') .addItem('Crear/verificar columnas de seguimiento', 'inicializarColumnas') .addToUi(); } /** Versión de ordenarHoja_ pensada para ejecutarse desde el menú de Sheets: además de * ordenar, limpia la caché y avisa con un mensaje cuando termina. Con ~19,700 filas puede * tardar unos segundos. */ function ordenarHojaManual_() { ordenarHoja_(); invalidarCache_(); SpreadsheetApp.getUi().alert('Listo: la hoja quedó ordenada por Sede → Servicio → Descripción.'); } function inicializarColumnas() { const sheet = getHoja_(); const lastCol = sheet.getLastColumn(); const encabezados = sheet.getRange(CONFIG.FILA_ENCABEZADO, 1, 1, lastCol).getValues()[0]; const nombresNuevos = [ COL_NUEVAS.FECHA_REVISION, COL_NUEVAS.TECNICO, COL_NUEVAS.OBSERVACIONES, ]; let col = lastCol; nombresNuevos.forEach(function (nombre) { if (encabezados.indexOf(nombre) === -1) { col++; sheet.getRange(CONFIG.FILA_ENCABEZADO, col).setValue(nombre); } }); SpreadsheetApp.getUi().alert('Columnas creadas/verificadas correctamente.'); } /** Devuelve el índice de la columna por nombre. Si no existe, la CREA automáticamente (autorreparable). */ function indiceColumnaNueva_(sheet, nombreColumna) { const lastCol = sheet.getLastColumn(); const encabezados = sheet.getRange(CONFIG.FILA_ENCABEZADO, 1, 1, lastCol).getValues()[0]; const idx = encabezados.indexOf(nombreColumna); if (idx !== -1) return idx + 1; const nuevaCol = lastCol + 1; sheet.getRange(CONFIG.FILA_ENCABEZADO, nuevaCol).setValue(nombreColumna); return nuevaCol; } /** Quita ceros a la izquierda de códigos puramente numéricos (029777 -> 29777), para que * no importe si el escáner o la etiqueta física trae ceros de más. Los códigos con letras * u otros caracteres se dejan intactos. */ function normalizarCodigo_(valor) { const str = (valor || '').toString().trim(); if (/^\d+$/.test(str)) return str.replace(/^0+(?=\d)/, ''); return str; } /* ================= CACHÉ (helpers genéricos, con chunking para objetos grandes) ================= */ const TAM_CHUNK = 90000; // bytes por entrada de caché (límite real de CacheService es 100KB) const CACHE_MAX_CHUNKS = 300; // seguro de tamaño, no debería alcanzarse /** Guarda cualquier valor serializable en la caché, partiéndolo en trozos si hace falta. */ function guardarJsonEnCache_(prefijo, valor) { const cache = CacheService.getScriptCache(); const json = JSON.stringify(valor); const chunks = []; for (let i = 0; i < json.length; i += TAM_CHUNK) chunks.push(json.substring(i, i + TAM_CHUNK)); if (chunks.length > CACHE_MAX_CHUNKS) return; const paquete = {}; chunks.forEach(function (c, i) { paquete[prefijo + '_' + i] = c; }); paquete[prefijo + '_n'] = String(chunks.length); try { cache.putAll(paquete, CONFIG.CACHE_TTL_SEGUNDOS); } catch (e) {} } /** Lee y reconstruye un valor previamente guardado con guardarJsonEnCache_. Devuelve null * si no hay caché válida (para que el llamador decida si reconstruir desde la hoja). */ function leerJsonDeCache_(prefijo) { const cache = CacheService.getScriptCache(); const numChunks = cache.get(prefijo + '_n'); if (!numChunks) return null; const claves = []; for (let i = 0; i < Number(numChunks); i++) claves.push(prefijo + '_' + i); const partes = cache.getAll(claves); let json = ''; for (let i = 0; i < claves.length; i++) { if (!partes[claves[i]]) return null; json += partes[claves[i]]; } try { return JSON.parse(json); } catch (e) { return null; } } function invalidarPrefijoCache_(prefijo) { const cache = CacheService.getScriptCache(); const numChunks = cache.get(prefijo + '_n'); const claves = [prefijo + '_n']; if (numChunks) { for (let i = 0; i < Number(numChunks); i++) claves.push(prefijo + '_' + i); } cache.removeAll(claves); } // v5: la vigencia se lee desde una fecha de alta REAL (columna E). const CACHE_PREFIJO = 'indice_v5'; // índice completo (para listar por sede/global) const CACHE_PREFIJO_CODIGOS = 'indice_codigos_v1'; // índice liviano: código normalizado -> fila const CACHE_ESTADISTICAS = 'estadisticas_v1'; // { totalEquipos, actualizados, porcentaje } // Posiciones dentro de cada arreglo del índice completo (más liviano que usar objetos) const IDX = { FILA: 0, COD_PAT: 1, COD_BAR: 2, DESC: 3, SEDE: 4, FECHA_ALTA: 5, MARCA: 6, MODELO: 7, SERIE: 8, SERVICIO: 9, ESTADO: 10, FECHA_REVISION: 11, TECNICO: 12, OBSERVACIONES: 13, }; /** Índice completo de todos los equipos (para listar por sede, listar global y como respaldo * del porcentaje). Es el más caro de construir; por eso buscarPorCodigo_ y las estadísticas * ya NO dependen de él en el camino normal. */ function obtenerIndice_() { const cacheado = leerJsonDeCache_(CACHE_PREFIJO); if (cacheado) return cacheado; const sheet = getHoja_(); const lastRow = sheet.getLastRow(); if (lastRow < 2) return []; // Las columnas de metadata (fecha/técnico/observación) se crean dinámicamente // al final de la hoja, así que hay que ubicarlas por nombre antes de leer. const colFecha = indiceColumnaNueva_(sheet, COL_NUEVAS.FECHA_REVISION); const colTecnico = indiceColumnaNueva_(sheet, COL_NUEVAS.TECNICO); const colObservaciones = indiceColumnaNueva_(sheet, COL_NUEVAS.OBSERVACIONES); const ultimaColumna = Math.max(COL.ESTADO, colFecha, colTecnico, colObservaciones); const datos = sheet.getRange(2, 1, lastRow - 1, ultimaColumna).getValues(); const indice = []; for (let i = 0; i < datos.length; i++) { const fila = datos[i]; if (fila[COL.DESCRIPCION - 1]) { indice.push([ i + 2, fila[COL.CODIGO_PATRIMONIAL - 1], fila[COL.CODIGO_BARRAS - 1], fila[COL.DESCRIPCION - 1], fila[COL.SEDE - 1], fila[COL.FECHA_ALTA - 1] ? new Date(fila[COL.FECHA_ALTA - 1]).getTime() : '', fila[COL.MARCA - 1], fila[COL.MODELO - 1], fila[COL.SERIE - 1], fila[COL.SERVICIO - 1], fila[COL.ESTADO - 1], fila[colFecha - 1] ? new Date(fila[colFecha - 1]).getTime() : '', fila[colTecnico - 1] || '', fila[colObservaciones - 1] || '', ]); } } guardarJsonEnCache_(CACHE_PREFIJO, indice); return indice; } /** Índice liviano código -> número de fila (solo columnas A:C). Mucho más barato de construir * y de leer que el índice completo, así que buscar por código nunca tiene que esperar a que * se reconstruya el índice pesado de 19,700 filas. */ function obtenerIndiceCodigos_() { const cacheado = leerJsonDeCache_(CACHE_PREFIJO_CODIGOS); if (cacheado) return cacheado; const sheet = getHoja_(); const lastRow = sheet.getLastRow(); if (lastRow < 2) return {}; const datos = sheet.getRange(2, 1, lastRow - 1, COL.DESCRIPCION).getValues(); // solo A:C const mapa = {}; for (let i = 0; i < datos.length; i++) { const fila = datos[i]; if (!fila[COL.DESCRIPCION - 1]) continue; const filaNum = i + 2; const codPat = normalizarCodigo_(fila[COL.CODIGO_PATRIMONIAL - 1]); const codBar = normalizarCodigo_(fila[COL.CODIGO_BARRAS - 1]); if (codPat) mapa[codPat] = filaNum; if (codBar) mapa[codBar] = filaNum; } guardarJsonEnCache_(CACHE_PREFIJO_CODIGOS, mapa); return mapa; } /** Limpieza TOTAL: se usa desde el botón "Actualizar sistema" y desde el menú de la hoja, * cuando se quiere forzar que todo se reconstruya de cero. */ function invalidarCache_() { invalidarPrefijoCache_(CACHE_PREFIJO); invalidarPrefijoCache_(CACHE_PREFIJO_CODIGOS); CacheService.getScriptCache().remove(CACHE_ESTADISTICAS); } /** Se usa tras registrar un equipo nuevo: como la hoja se reordena, los números de fila de * todos los equipos pueden cambiar, así que el índice completo y el de códigos ya no sirven. * Las estadísticas NO se invalidan aquí: se ajustan aparte, en memoria, sin recalcular todo. */ function invalidarIndicesPorReordenamiento_() { invalidarPrefijoCache_(CACHE_PREFIJO); invalidarPrefijoCache_(CACHE_PREFIJO_CODIGOS); } /* ================= VIGENCIA (edad del equipo, calculada desde la fecha de alta REAL) ================= */ /** Años transcurridos (decimal) desde la fecha de alta del equipo */ function calcularVigenciaDesdeAlta_(fechaAltaTimestamp) { if (!fechaAltaTimestamp) return null; const diffMs = Date.now() - fechaAltaTimestamp; if (diffMs < 0) return 0; return diffMs / MS_POR_ANIO; } /** Convierte años (decimal) en texto "X AÑOS Y MESES" — siempre con ambas partes, en mayúscula, * igual al formato que ya usa la columna F de la hoja (ej: "13 AÑOS 2 MESES"). */ function formatVigenciaTexto_(aniosDecimal) { if (aniosDecimal === null || aniosDecimal === undefined) return 'S/D'; let anios = Math.floor(aniosDecimal); let meses = Math.round((aniosDecimal - anios) * 12); if (meses === 12) { meses = 0; anios += 1; } return anios + ' AÑOS ' + meses + ' MESES'; } /* ================= VIDA ÚTIL (pestaña "VIDA UTIL") ================= */ const CACHE_VIDA_UTIL = 'vida_util_v1'; /** Devuelve un objeto { DENOMINACION_EN_MAYUSCULAS: vidaUtilEnAños } */ function obtenerMapaVidaUtil_() { const cache = CacheService.getScriptCache(); const guardado = cache.get(CACHE_VIDA_UTIL); if (guardado) { try { return JSON.parse(guardado); } catch (e) {} } const ss = SpreadsheetApp.getActiveSpreadsheet(); const hoja = ss.getSheetByName(CONFIG.SHEET_NAME_VIDA_UTIL); if (!hoja) return {}; const lastRow = hoja.getLastRow(); if (lastRow < 2) return {}; const datos = hoja.getRange(2, 1, lastRow - 1, COL_VIDA_UTIL.VIDA_UTIL).getValues(); const mapa = {}; datos.forEach(function (fila) { const denominacion = (fila[COL_VIDA_UTIL.DENOMINACION - 1] || '').toString().trim().toUpperCase(); const vidaUtil = fila[COL_VIDA_UTIL.VIDA_UTIL - 1]; if (denominacion && vidaUtil !== '' && vidaUtil !== null) { mapa[denominacion] = vidaUtil; } }); try { cache.put(CACHE_VIDA_UTIL, JSON.stringify(mapa), CONFIG.CACHE_TTL_SEGUNDOS); } catch (e) { // si el catálogo es muy grande para cachear, simplemente no se cachea } return mapa; } /** Busca la vida útil de un equipo comparando su descripción contra el catálogo */ function buscarVidaUtil_(descripcionEquipo) { if (!descripcionEquipo) return null; const mapa = obtenerMapaVidaUtil_(); const desc = descripcionEquipo.toString().trim().toUpperCase(); // 1. Coincidencia exacta if (mapa[desc] !== undefined) return mapa[desc]; // 2. Coincidencia parcial (la descripción contiene la denominación, o viceversa) let mejorCoincidencia = null; let mejorLongitud = 0; Object.keys(mapa).forEach(function (denominacion) { if (desc.indexOf(denominacion) !== -1 || denominacion.indexOf(desc) !== -1) { if (denominacion.length > mejorLongitud) { mejorLongitud = denominacion.length; mejorCoincidencia = mapa[denominacion]; } } }); return mejorCoincidencia; } /* ================= ENDPOINTS ================= */ function doGet(e) { try { const accion = e.parameter.accion; let resultado; if (accion === 'buscarPorCodigo') { resultado = buscarPorCodigo_(e.parameter.codigo); } else if (accion === 'listarSedes') { resultado = listarSedes_(); } else if (accion === 'listarDependencias') { resultado = listarServicios_(e.parameter.sede); } else if (accion === 'listarEquipos') { resultado = listarEquipos_(e.parameter.sede, e.parameter.dependencia); } else if (accion === 'listarSedeCompleta') { resultado = listarSedeCompleta_(e.parameter.sede, e.parameter.dependencia); } else if (accion === 'listarTodosLosEquiposLiviano') { resultado = listarTodosLosEquiposLiviano_(); } else if (accion === 'obtenerEquipoPorFila') { resultado = obtenerEquipoPorFila_(e.parameter.fila); } else if (accion === 'estadisticas') { resultado = obtenerEstadisticasGlobales_(); } else if (accion === 'limpiarCache') { invalidarCache_(); resultado = { ok: true, mensaje: 'Caché limpiada.' }; } else { resultado = { error: 'Acción no reconocida' }; } return respuestaJson_(resultado); } catch (err) { return respuestaJson_({ error: err.message }); } } function doPost(e) { try { const datos = JSON.parse(e.postData.contents); let resultado; if (datos.accion === 'registrar') { resultado = registrarEquipo_(datos); } else { resultado = actualizarEstado_(datos); } return respuestaJson_(resultado); } catch (err) { return respuestaJson_({ error: err.message }); } } function respuestaJson_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } /* ================= BÚSQUEDA ================= */ /** Busca un equipo por código de barras o patrimonial. Usa el índice liviano (solo código -> * fila) y después lee ESA UNA fila directo de la hoja — nunca carga ni espera el índice * pesado de 19,700 filas, así que es rápida incluso justo después de una actualización. */ function buscarPorCodigo_(codigo) { if (!codigo) return { error: 'Código vacío' }; const codigoStr = codigo.toString().trim(); const codigoNormalizado = normalizarCodigo_(codigoStr); const mapaCodigos = obtenerIndiceCodigos_(); const fila = mapaCodigos[codigoNormalizado] || mapaCodigos[codigoStr]; if (!fila) { return { error: 'No se encontró ningún equipo con ese código en la base de datos.', noEncontrado: true, codigo: codigoStr }; } return obtenerEquipoPorFila_(fila); } function listarSedes_() { const indice = obtenerIndice_(); const sedes = new Set(); indice.forEach(function (eq) { const sede = (eq[IDX.SEDE] || '').toString().trim(); if (sede) sedes.add(sede); }); return { sedes: Array.from(sedes).sort() }; } function listarServicios_(sede) { const indice = obtenerIndice_(); const servicios = new Set(); indice.forEach(function (eq) { if ((eq[IDX.SEDE] || '').toString().trim() === sede) { const servicio = (eq[IDX.SERVICIO] || '').toString().trim(); if (servicio) servicios.add(servicio); } }); return { dependencias: Array.from(servicios).sort() }; } function listarEquipos_(sede, servicio) { const indice = obtenerIndice_(); const equipos = indice .filter(function (eq) { const coincideSede = (eq[IDX.SEDE] || '').toString().trim() === sede; const coincideServicio = !servicio || (eq[IDX.SERVICIO] || '').toString().trim() === servicio; return coincideSede && coincideServicio; }) .map(equipoAObjetoRespuesta_); return { equipos: equipos }; } /** Trae servicios y equipos de una sede en UNA sola llamada (evita dos viajes al servidor) */ function listarSedeCompleta_(sede, servicio) { const indice = obtenerIndice_(); const servicios = new Set(); const equipos = []; indice.forEach(function (eq) { if ((eq[IDX.SEDE] || '').toString().trim() !== sede) return; const servicioEq = (eq[IDX.SERVICIO] || '').toString().trim(); if (servicioEq) servicios.add(servicioEq); if (!servicio || servicioEq === servicio) { equipos.push(equipoAObjetoRespuesta_(eq)); } }); return { dependencias: Array.from(servicios).sort(), equipos: equipos }; } /** Todos los equipos de la base, SIN filtrar por sede y SIN calcular vigencia/vida útil — * esa parte es la más pesada de equipoAObjetoRespuesta_ (compara cada equipo contra el * catálogo "VIDA UTIL"), y no hace falta para poblar un buscador. Solo trae lo necesario * para mostrar y filtrar resultados: descripción, marca, modelo, serie, sede y estado. * Cuando el técnico elige uno, se pide el detalle completo con obtenerEquipoPorFila_. */ function listarTodosLosEquiposLiviano_() { const indice = obtenerIndice_(); const equipos = indice.map(function (eq) { return { fila: eq[IDX.FILA], codigoPatrimonial: eq[IDX.COD_PAT], codigoBarras: eq[IDX.COD_BAR], descripcion: eq[IDX.DESC], sede: eq[IDX.SEDE], dependencia: eq[IDX.SERVICIO], marca: eq[IDX.MARCA], modelo: eq[IDX.MODELO], serie: eq[IDX.SERIE], estadoActual: eq[IDX.ESTADO], }; }); return { equipos: equipos }; } /** Detalle completo (vigencia, vida útil, última actualización) de UN equipo puntual, * ubicado por su número de fila. Lee esa única fila directo de la hoja: no depende del * índice completo ni de su estado de caché, así que siempre es rápido y siempre trae el * dato más fresco (útil justo después de buscar por código o de actualizar el equipo). */ function obtenerEquipoPorFila_(filaParam) { const fila = Number(filaParam); if (!fila) return { error: 'Fila inválida.' }; const sheet = getHoja_(); if (fila < 2 || fila > sheet.getLastRow()) { return { error: 'No se encontró el equipo. Prueba "Actualizar sistema" y vuelve a intentar.' }; } const colFecha = indiceColumnaNueva_(sheet, COL_NUEVAS.FECHA_REVISION); const colTecnico = indiceColumnaNueva_(sheet, COL_NUEVAS.TECNICO); const colObservaciones = indiceColumnaNueva_(sheet, COL_NUEVAS.OBSERVACIONES); const ultimaColumna = Math.max(COL.ESTADO, colFecha, colTecnico, colObservaciones); const valores = sheet.getRange(fila, 1, 1, ultimaColumna).getValues()[0]; if (!valores[COL.DESCRIPCION - 1]) { return { error: 'No se encontró el equipo. Prueba "Actualizar sistema" y vuelve a intentar.' }; } const eq = [ fila, valores[COL.CODIGO_PATRIMONIAL - 1], valores[COL.CODIGO_BARRAS - 1], valores[COL.DESCRIPCION - 1], valores[COL.SEDE - 1], valores[COL.FECHA_ALTA - 1] ? new Date(valores[COL.FECHA_ALTA - 1]).getTime() : '', valores[COL.MARCA - 1], valores[COL.MODELO - 1], valores[COL.SERIE - 1], valores[COL.SERVICIO - 1], valores[COL.ESTADO - 1], valores[colFecha - 1] ? new Date(valores[colFecha - 1]).getTime() : '', valores[colTecnico - 1] || '', valores[colObservaciones - 1] || '', ]; return equipoAObjetoRespuesta_(eq); } /** % de equipos que ya tienen al menos una actualización registrada. Vive en su propia * caché chica: la inmensa mayoría de las veces esto es solo un cache.get() de un objeto * pequeño, sin tocar el índice pesado de 19,700 filas para nada. */ function obtenerEstadisticasGlobales_() { const cache = CacheService.getScriptCache(); const guardado = cache.get(CACHE_ESTADISTICAS); if (guardado) { try { return JSON.parse(guardado); } catch (e) {} } const indice = obtenerIndice_(); const total = indice.length; let actualizados = 0; indice.forEach(function (eq) { if (eq[IDX.FECHA_REVISION]) actualizados++; }); const porcentaje = total > 0 ? Math.round((actualizados / total) * 1000) / 10 : 0; const resultado = { totalEquipos: total, actualizados: actualizados, porcentaje: porcentaje }; try { cache.put(CACHE_ESTADISTICAS, JSON.stringify(resultado), CONFIG.CACHE_TTL_SEGUNDOS); } catch (e) {} return resultado; } /** Se llama justo después de la PRIMERA actualización de un equipo (antes no tenía ninguna): * suma 1 al contador de "actualizados" en la caché chica, sin releer toda la hoja.
